function pythonToJS(code) {

  let lines = code.split('\n'), js = '', indentStack = [0], declared = new Set();

  for (let i = 0; i < lines.length; i++) {

    let line = lines[i], stripped = line.trimStart();

    if (!stripped || stripped.startsWith('#')) continue;

    let indent = line.length - stripped.length;

    while (indent < indentStack[indentStack.length-1]) { indentStack.pop(); js += '}\n'; }

    let out = stripped;

    out = out.replace(/^import numpy as np$/, 'const np = null;');

    out = out.replace(/^import pandas as pd$/, 'const pd = null;');

    out = out.replace(/^import .*$/, '');

    out = out.replace(/^for (\w+) in range\((\S+)\)\s*:/, (_, v, n) => `for(let ${v}=0;${v}<${n};${v}++){`);

    out = out.replace(/^for (\w+) in range\((\S+),\s*(\S+)\)\s*:/, (_, v, a, b) => `for(let ${v}=${a};${v}<${b};${v}++){`);

    out = out.replace(/^for (\w+) in range\((\S+),\s*(\S+),\s*(-?\d+)\)\s*:/, (_, v, a, b, s) => parseInt(s)>0?`for(let ${v}=${a};${v}<${b};${v}+=${s}){`:`for(let ${v}=${a};${v}>${b};${v}+=(${s})){`);

    out = out.replace(/^for (\w+) in (\w+)\s*:/, (_, v, arr) => `for(const ${v} of ${arr}){`);

    out = out.replace(/^for (\w+) in \[([^\]]+)\]\s*:/, (_, v, arr) => `for(const ${v} of [${arr}]){`);

    out = out.replace(/^while (.+)\s*:/, (_, c) => `while(${c.replace(/\band\b/g,'&&').replace(/\bor\b/g,'||')}){`);

    out = out.replace(/^if (.+)\s*:/, (_, c) => `if(${c.replace(/\band\b/g,'&&').replace(/\bor\b/g,'||').replace(/\bnot\b/g,'!')}){`);

    out = out.replace(/^elif (.+)\s*:/, (_, c) => `}else if(${c.replace(/\band\b/g,'&&').replace(/\bor\b/g,'||')}){`);

    out = out.replace(/^else\s*:/, '}else{');

    out = out.replace(/^def (\w+)\(([^)]*)\)\s*:/, (_, n, a) => `function ${n}(${a}){`);

    out = out.replace(/^return (.+)/, 'return $1;');

    out = out.replace(/^return$/, 'return;');

    out = out.replace(/^break$/, 'break;');

    out = out.replace(/^continue$/, 'continue;');

    out = out.replace(/^raise (\w+)\((.+)\)/, (_, e, m) => `throw new Error(${m});`);

    out = out.replace(/^try\s*:/, 'try{');

    out = out.replace(/^except.*:/, '}catch(_e){');

    out = out.replace(/^finally\s*:/, '}finally{');

    out = out.replace(/print\(f"([^"]+)"\)/g, (_, t) => `print(\`${t.replace(/\{([^}]+)\}/g,'${$1}')}\`);`);

    out = out.replace(/print\(f'([^']+)'\)/g, (_, t) => `print(\`${t.replace(/\{([^}]+)\}/g,'${$1}')}\`);`);

    out = out.replace(/print\((.+)\)/g, (_, a) => `print(${a});`);

    out = out.replace(/(\w+) \+= (.+)/, '$1 += $2;');

    out = out.replace(/(\w+) -= (.+)/, '$1 -= $2;');

    out = out.replace(/(\w+) \*= (.+)/, '$1 *= $2;');

    out = out.replace(/\bTrue\b/g,'true').replace(/\bFalse\b/g,'false').replace(/\bNone\b/g,'null');

    out = out.replace(/\blen\((\w+)\)/g,'$1.length');

    out = out.replace(/\bstr\(([^)]+)\)/g,'String($1)');

    out = out.replace(/\bint\(([^)]+)\)/g,'parseInt($1)');

    out = out.replace(/\bfloat\(([^)]+)\)/g,'parseFloat($1)');

    out = out.replace(/\babs\(([^)]+)\)/g,'Math.abs($1)');

    out = out.replace(/\bmin\(([^)]+)\)/g,'_min($1)');

    out = out.replace(/\bmax\(([^)]+)\)/g,'_max($1)');

    out = out.replace(/\bsum\(([^)]+)\)/g,'_sum($1)');

    out = out.replace(/\bsorted\(([^)]+)\)/g,'_sorted($1)');

    out = out.replace(/\brange\(([^)]+)\)/g,'_range($1)');

    out = out.replace(/\.append\(/g,'.push(');

    out = out.replace(/\.strip\(\)/g,'.trim()');

    out = out.replace(/\.lower\(\)/g,'.toLowerCase()');

    out = out.replace(/\.upper\(\)/g,'.toUpperCase()');

    out = out.replace(/\.split\(\)/g,".split(' ')");

    out = out.replace(/\.split\(([^)]+)\)/g,'.split($1)');

    out = out.replace(/\.replace\(([^,]+),\s*([^)]+)\)/g,'.replace(new RegExp($1,"g"),$2)');

    out = out.replace(/\.count\(([^)]+)\)/g,(_, c) => `.split(${c}).length-1`);

    out = out.replace(/\.startswith\(/g,'.startsWith(');

    out = out.replace(/\.endswith\(/g,'.endsWith(');

    out = out.replace(/\.title\(\)/g,".replace(/\\b\\w/g,c=>c.toUpperCase())");

    out = out.replace(/\.find\(([^)]+)\)/g,'.indexOf($1)');

    out = out.replace(/\.get\(([^,]+),\s*([^)]+)\)/g,'?.[String($1)] ?? $2');

    const am = out.match(/^(\w+) = (.+)/);

    if (am && !out.startsWith('let ') && !out.startsWith('const ') && !out.startsWith('for') && !out.startsWith('while') && !out.startsWith('if') && !out.startsWith('function') && !out.startsWith('}') && !out.startsWith('return') && !out.startsWith('throw')) {

      if (!declared.has(am[1])) { out = 'let ' + out; declared.add(am[1]); }

    }

    if (out.endsWith('{')) indentStack.push(indent + 1);

    js += out + '\n';

  }

  while (indentStack.length > 1) { js += '}\n'; indentStack.pop(); }

  return js;

}


function runCode(i) {

  const ex = curriculum[state.currentTopic].exercises[i];

  const code = document.getElementById('code-area').value.trim();

  const outBox = document.getElementById('output-box');

  if (!code || code.startsWith('#')) { outBox.className='output-box'; outBox.textContent='Write some code and click Run!'; return; }

  let output = [];

  const fakePrint = (...args) => output.push(args.map(a => a===null?'None':a===true?'True':a===false?'False':Array.isArray(a)?'['+a.join(', ')+']':String(a)).join(' '));

  const helpers = `

    function _min(){const a=arguments.length===1&&Array.isArray(arguments[0])?arguments[0]:[...arguments];return Math.min(...a);}

    function _max(){const a=arguments.length===1&&Array.isArray(arguments[0])?arguments[0]:[...arguments];return Math.max(...a);}

    function _sum(a){return a.reduce((s,x)=>s+x,0);}

    function _sorted(a){const c=[...a];c.sort((x,y)=>x>y?1:-1);return c;}

    function _range(a,b,s){const r=[];if(b===undefined){for(let i=0;i<a;i++)r.push(i);}else{const st=s||1;if(st>0){for(let i=a;i<b;i+=st)r.push(i);}else{for(let i=a;i>b;i+=st)r.push(i);}}return r;}

    function zip(a,b){return a.map((x,i)=>[x,b[i]]);}

  `;

  try {

    const jsCode = pythonToJS(code);

    const fn = new Function('print', helpers + jsCode);

    fn(fakePrint);

    const result = output.join('\n');

    const ok = ex.check(result);

    outBox.textContent = (result || '(no output printed)') + (ok ? '\n\n✓ Correct!' : '');

    outBox.className = 'output-box error';

  }

}
